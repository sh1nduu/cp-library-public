```rust
use ac_library::{Modulus, StaticModInt};

#[derive(Debug)]
#[allow(dead_code)]
struct Combination<T> {
    max: usize,
    fact: Vec<T>,
    inv: Vec<T>,
    finv: Vec<T>,
}

#[allow(dead_code)]
impl<M: Modulus> Combination<StaticModInt<M>> {
    fn new(n: usize) -> Self {
        let mut fact = vec![StaticModInt::<M>::new(1); n + 1];
        let mut inv = vec![StaticModInt::<M>::new(1); n + 1];
        let mut finv = vec![StaticModInt::<M>::new(1); n + 1];
        let modulus = StaticModInt::<M>::modulus() as usize;

        for i in 2..=n {
            fact[i] = fact[i - 1] * i;
            inv[i] = -inv[modulus % i] * (modulus / i);
            finv[i] = finv[i - 1] * inv[i];
        }

        Self { max: n, fact, inv, finv }
    }

    fn calc(&self, n: usize, k: usize) -> StaticModInt<M> {
        self.fact[n] * self.finv[k] * self.finv[n - k]
    }
}
```
