```rust
fn align(a: &HashSet<(i64, i64)>) -> HashSet<(i64, i64)> {
    let &(mi, mj) = a.iter().min().unwrap();
    a.iter().map(|&(i, j)| (i - mi, j - mj)).collect()
}
```

```rust
fn rot_cw(a: &Vec<Vec<char>>) -> Vec<Vec<char>> {
    let n = a.len();
    let mut b = vec![vec!['.'; n]; n];
    for i in 0..n {
        for j in 0..n {
            b[j][n - 1 - i] = a[i][j];
        }
    }
    b
}
```
