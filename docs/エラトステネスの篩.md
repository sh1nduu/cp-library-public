## Code 1

```rust
let mut is_prime = vec![true; n + 1];
is_prime[0] = false;
is_prime[1] = false;
for i in 2..=n {
    if is_prime[i] {
        for j in (i * i..=n).step_by(i) {
            is_prime[j] = false;
        }
    }
}
```

let primes: Vec<usize> = (2..=n).filter(|&i| is_prime[i]).collect();
