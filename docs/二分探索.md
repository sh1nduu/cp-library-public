## Code 1

```rust
fn binary_search_answer<F>(mut ok: i64, mut ng: i64, mut is_ok: F) -> i64
where
    F: FnMut(i64) -> bool,
{
    while ok.abs_diff(ng) > 1 {
        let mid = ok.midpoint(ng);
        if is_ok(mid) {
            ok = mid;
        } else {
            ng = mid;
        }
    }
    ok
}
```

```rust
fn main() {
    let n: i64 = 1_000_000_000;
    // x^2 が n を超える最小の整数 x をもとめる.
    let ans = binary_search_answer(1_000_000_000, 0, |x| x * x >= n);
    println!("{ans}");
}
```

## Code 3

```rust
fn binary_search_real<F>(mut ok: f64, mut ng: f64, iterations: usize, mut is_ok: F) -> f64
where
    F: FnMut(f64) -> bool,
{
    for _ in 0..iterations {
        let mid = (ok + ng) / 2.0;
        if is_ok(mid) {
            ok = mid;
        } else {
            ng = mid;
        }
    }
    ok
}
```


## Code 4

```rust
fn main() {
    // 例: x^2 >= 2 を満たす最小の x を求める
    let ans = binary_search_real(2.0, 0.0, 80, |x| x * x >= 2.0);
    println!("{:.10}", ans); // ≒ 1.4142135623
}
```
