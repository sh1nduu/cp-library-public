use std::collections::BTreeMap;

fn prime_factors(mut x: usize) -> BTreeMap<usize, usize> {
    let mut map = BTreeMap::new();
    let mut i = 2;
    while i * i <= x {
        while x % i == 0 {
            x /= i;
            *map.entry(i).or_insert(0) += 1;
        }
        i += 1;
    }
    if x > 1 {
        *map.entry(x).or_insert(0) += 1;
    }
    map
}
```

fn sieve_spf(n: usize) -> Vec<usize> {
    let mut spf = vec![0; n + 1];
    for i in 2..=n {
        if spf[i] == 0 {
            spf[i] = i;
            for j in (i * i..=n).step_by(i) {
                if spf[j] == 0 {
                    spf[j] = i;
                }
            }
        }
    }
    spf
}
```

use std::collections::BTreeMap;

fn prime_factors_spf(mut x: usize, spf: &[usize]) -> BTreeMap<usize, usize> {
    let mut map = BTreeMap::new();
    while x > 1 {
        let p = spf[x];
        *map.entry(p).or_insert(0) += 1;
        x /= p;
    }
    map
}
```

fn divisors_from_factors(factors: &[(usize, usize)]) -> Vec<usize> {
    let mut res = vec![1usize];

    for &(p, exp) in factors {
        let s = res.len();
        for i in 0..s {
            let mut v = 1usize;
            for _ in 0..exp {
                v *= p;
                res.push(res[i] * v);
            }
        }
    }

    res.sort_unstable();
    res
}
```
