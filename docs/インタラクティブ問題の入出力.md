use std::io::{BufRead, BufReader, Write, stdout};
use proconio::{input, source::line::LineSource};

fn query<R: BufRead>(x: usize, src: &mut LineSource<R>) -> usize {
    println!("? {}", x);
    stdout().flush().unwrap();
    input! { from src, y: usize }
    y
}

fn main() {
    let mut stdin = LineSource::new(BufReader::new(std::io::stdin()));

    input! {
        from &mut stdin,
        n: usize,
    }

    let y = query(1, &mut stdin);
    // ...
}
